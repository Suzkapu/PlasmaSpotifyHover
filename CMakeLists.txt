cmake_minimum_required(VERSION 3.24)
project(PlasmaSpotifyHover LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
set(PLASMA_SKIP_APPLET_INSTALL ON)
set(KDE_INSTALL_USE_QT_SYS_PATHS OFF)

find_package(ECM 6.30 REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH "${ECM_MODULE_PATH}")
include(KDEInstallDirs)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(ECMQtDeclareLoggingCategory)

find_package(Qt6 6.11 REQUIRED COMPONENTS Core Quick DBus)
find_package(KF6 6.30 REQUIRED COMPONENTS Config CoreAddons GlobalAccel I18n KIO Notifications Service WindowSystem XmlGui)
find_package(Plasma 6.8.2 EXACT REQUIRED)
find_package(PlasmaActivities 6.8 REQUIRED)
find_package(PlasmaActivitiesStats 6.8 REQUIRED)
find_package(KSysGuard REQUIRED)
find_package(LibNotificationManager 6.8 REQUIRED)

add_subdirectory(applets/taskmanager)
